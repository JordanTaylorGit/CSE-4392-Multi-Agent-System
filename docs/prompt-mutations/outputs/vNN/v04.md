The following refactoring uses stateless, shared state objects. `SeasonSwitch.state` remains the single source of truth, and the state map is consulted on every delegated operation. Invalid state values are handled by a silent no-op state.

The original `main` method is preserved verbatim.

java

```java
package cse4392;

import javax.swing.Timer;
import java.awt.event.ActionEvent;
import java.awt.event.ActionListener;
import java.util.Map;
import java.util.concurrent.CountDownLatch;

interface SeasonState {
   void cool(SeasonSwitch context);
   void heat(SeasonSwitch context);
   void off(SeasonSwitch context);
   void timerOff(SeasonSwitch context);
   void roomActionPerformed(SeasonSwitch context);
}

class OffState implements SeasonState {
   public void cool(SeasonSwitch context) {
      context.ac.on();
      context.timer.start();
      SeasonSwitch.state=SeasonSwitch.COOL;
      System.out.println("Cooling started. RmTemp="+context.room.temp);
   }

   public void heat(SeasonSwitch context) {
      context.heater.on();
      context.timer.start();
      SeasonSwitch.state=SeasonSwitch.HEAT;
      System.out.println("Heating started. RmTemp="+context.room.temp);
   }

   public void off(SeasonSwitch context) {
   }

   public void timerOff(SeasonSwitch context) {
   }

   public void roomActionPerformed(SeasonSwitch context) {
   }
}

class CoolState implements SeasonState {
   public void cool(SeasonSwitch context) {
   }

   public void heat(SeasonSwitch context) {
      context.ac.off();
      context.heater.on();
      context.timer.start();
      SeasonSwitch.state=SeasonSwitch.HEAT;
      System.out.println("Heating started. RmTemp="+context.room.temp);
   }

   public void off(SeasonSwitch context) {
      context.ac.off();
      context.timer.stop();
      SeasonSwitch.state=SeasonSwitch.OFF;
      System.out.println("Cooling stopped. RmTemp="+context.room.temp);
   }

   public void timerOff(SeasonSwitch context) {
      if (context.room.temp>context.coolTemp) {
         context.ac.on();
         System.out.println("Cooling started. RmTemp="+context.room.temp);
      } else {
         context.ac.off();
         System.out.println("Cooling stopped. RmTemp="+context.room.temp);
      }
   }

   public void roomActionPerformed(SeasonSwitch context) {
      if (context.ac.isOn) context.room.temp--; else context.room.temp++;
   }
}

class HeatState implements SeasonState {
   public void cool(SeasonSwitch context) {
      context.heater.off();
      context.ac.on();
      context.timer.start();
      SeasonSwitch.state=SeasonSwitch.COOL;
      System.out.println("Cooling started. RmTemp="+context.room.temp);
   }

   public void heat(SeasonSwitch context) {
   }

   public void off(SeasonSwitch context) {
      context.heater.off();
      context.timer.stop();
      SeasonSwitch.state=SeasonSwitch.OFF;
      System.out.println("Heating stopped. RmTemp="+context.room.temp);
   }

   public void timerOff(SeasonSwitch context) {
      if (context.room.temp<context.heatTemp) {
         context.heater.on();
         System.out.println("Heating started. RmTemp="+context.room.temp);
      } else {
         context.heater.off();
         System.out.println("Heating stopped. RmTemp="+context.room.temp);
      }
   }

   public void roomActionPerformed(SeasonSwitch context) {
      if (context.heater.isOn) context.room.temp++; else context.room.temp--;
   }
}

class NoOpState implements SeasonState {
   public void cool(SeasonSwitch context) {
   }

   public void heat(SeasonSwitch context) {
   }

   public void off(SeasonSwitch context) {
   }

   public void timerOff(SeasonSwitch context) {
   }

   public void roomActionPerformed(SeasonSwitch context) {
   }
}

public class SeasonSwitch implements ActionListener {
   static int state=0;
   static final int OFF=0, COOL=1, HEAT=2;
   AC ac=new AC();
   Heater heater=new Heater();
   Timer timer=new Timer(1200, this);
   Room room=new Room();
   int coolTemp=68;
   int heatTemp=77;

   private static final SeasonState OFF_STATE = new OffState();
   private static final SeasonState COOL_STATE = new CoolState();
   private static final SeasonState HEAT_STATE = new HeatState();
   private static final SeasonState NO_OP_STATE = new NoOpState();

   private static final Map<Integer, SeasonState> STATES = Map.of(
      OFF, OFF_STATE,
      COOL, COOL_STATE,
      HEAT, HEAT_STATE
   );

   private SeasonState getState() {
      return STATES.getOrDefault(state, NO_OP_STATE);
   }

   public void cool() {
      getState().cool(this);
   }

   public void heat() {
      getState().heat(this);
   }

   public void off() {
      getState().off(this);
   }

   public void actionPerformed(ActionEvent e) {
      timerOff();
   }

   public void timerOff() {
      getState().timerOff(this);
   }

   public static void main(String[] args) {
      System.out.println("Enter off, cool or heat to run. "+
         "Press Ctrl-c to exit.");
      SeasonSwitch s=new SeasonSwitch();
      s.room.timer.start();
      if (args.length==0) s.cool(); else
      if ("off".compareTo(args[0])==0) {
         s.off(); 
         System.out.println("System off. System exits.");
         System.exit(0);
      } else
      if ("cool".compareTo(args[0])==0) s.cool(); else
      if ("heat".compareTo(args[0])==0) s.heat(); else {
         System.out.println("Unknown command. System exits.");
         System.exit(1);
      }
      try {
         new CountDownLatch(5).await();
      } catch (InterruptedException e) {
         Thread.currentThread().interrupt();
         s.timer.stop();
         s.room.timer.stop();
      }
   }

   class AC {
      boolean isOn=false;
      public void on() { isOn=true; }
      public void off() { isOn=false; }
   }

   class Heater {
      boolean isOn=false;
      public void on() { isOn=true; }
      public void off() { isOn=false; }
   }

   class Room implements ActionListener {
      public int temp=70;
      Timer timer=new Timer(400, this);

      public void actionPerformed(ActionEvent ae) {
         getState().roomActionPerformed(SeasonSwitch.this);
      }
   }
}
```

### Design notes

- Context: `SeasonSwitch` delegates all five state-dependent operations to `getState()`.
- State lookup: `getState()` uses `STATES.getOrDefault(state, NO_OP_STATE)` every time. Direct assignments to `SeasonSwitch.state`, including invalid values, are respected.
- Concrete states: `OffState`, `CoolState`, and `HeatState` own their transitions, device operations, timer operations, and console messages.
- Invalid state: `NoOpState` silently ignores every operation.
- Room: Temperature drift is delegated to the active state, eliminating the original state switch from `Room`.
- Shared instances: All state objects are stateless and shared through `static final` references.

Coverage note: The existing `if` conditions for temperature thresholds and device status are preserved. No additional conditional logic is introduced. The refactoring preserves the original behavior and side-effect ordering.
