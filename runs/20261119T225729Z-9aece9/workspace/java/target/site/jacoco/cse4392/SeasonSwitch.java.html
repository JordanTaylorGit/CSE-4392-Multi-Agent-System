package cse4392;

import javax.swing.Timer;
import java.awt.event.ActionEvent;
import java.awt.event.ActionListener;
import java.util.concurrent.CountDownLatch;
import java.util.Map;

public class SeasonSwitch implements ActionListener {
    static int state=0;
    static final int OFF=0, COOL=1, HEAT=2;

    AC ac=new AC();
    Heater heater=new Heater();
    Timer timer=new Timer(1200, this);
    Room room=new Room();
    int coolTemp=68;
    int heatTemp=77;

    private static final SeasonState OFF_STATE = new OffState();
    private static final SeasonState COOL_STATE = new CoolState();
    private static final SeasonState HEAT_STATE = new HeatState();
    private static final SeasonState NO_OP_STATE = new NoOpState();

    private static final Map<Integer, SeasonState> STATES = Map.of(
        OFF, OFF_STATE,
        COOL, COOL_STATE,
        HEAT, HEAT_STATE
    );

    SeasonState getCurrentState() {
        return STATES.getOrDefault(state, NO_OP_STATE);
    }

    public void cool() {
        getCurrentState().cool(this);
    }

    public void heat() {
        getCurrentState().heat(this);
    }

    public void off() {
        getCurrentState().off(this);
    }

    public void actionPerformed(ActionEvent e) {
        timerOff();
    }

    public void timerOff() {
        getCurrentState().timerOff(this);
    }

    public static void main(String[] args) {
        System.out.println("Enter off, cool or heat to run. "+
           "Press Ctrl-c to exit.");
        SeasonSwitch s=new SeasonSwitch();
        s.room.timer.start();
        if (args.length==0) s.cool(); else
        if ("off".compareTo(args[0])==0) {
           s.off(); 
           System.out.println("System off. System exits.");
           System.exit(0);
        } else
        if ("cool".compareTo(args[0])==0) s.cool(); else
        if ("heat".compareTo(args[0])==0) s.heat(); else {
           System.out.println("Unknown command. System exits.");
           System.exit(1);
        }
        try {
           new CountDownLatch(5).await();
        } catch (InterruptedException e) {
           Thread.currentThread().interrupt();
           s.timer.stop();
           s.room.timer.stop();
        }
    }

    class AC {
        boolean isOn=false;

        public void on() {
            isOn=true;
        }

        public void off() {
            isOn=false;
        }
    }

    class Heater {
        boolean isOn=false;

        public void on() {
            isOn=true;
        }

        public void off() {
            isOn=false;
        }
    }

    class Room implements ActionListener {
        public int temp=70;
        Timer timer=new Timer(400, this);

        public void actionPerformed(ActionEvent ae) {
            getCurrentState().roomActionPerformed(SeasonSwitch.this);
        }
    }
}
